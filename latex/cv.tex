    \section{Programming Projects}
        \begin{itemize}
            \item
                \href{https://github.com/ryanmaguire/libtmpl/}{\texttt{libtmpl}}
                \begin{itemize}
                    \item
                        Nearly complete implementation of \texttt{libm}, the
                        C standard mathematical library, with additional
                        features such as vector geometry, algebra, knot theory,
                        Fourier transforms, and complex numbers.
                \end{itemize}
            \item
                \href{%
                    https://github.com/NASA-Planetary-Science/rss_ringoccs/%
                }{\texttt{rss\_ringoccs}}
                \begin{itemize}
                    \item
                        Suite of code for processing and analyzing radio
                        science data from NASA's Cassini mission. It
                        numerically inverts data from occultation
                        observations and can be used to study the rings of
                        Saturn. Recent improvements allow the code to be used
                        for the rings of Uranus as well.
                \end{itemize}
            \item
                \href{%
                    https://github.com/ryanmaguire/Mathematics-and-Physics/%
                }{\texttt{Mathematics-and-Physics}}
                \begin{itemize}
                    \item
                        Notes for the courses I taught at Dartmouth and MIT,
                        and for the courses I assisted at Dartmouth and
                        UMass Lowell.
                \end{itemize}
            \item
                \href{%
                    https://github.com/ryanmaguire/barnsley_fern/%
                }{\texttt{barnsley\_fern}}
                \begin{itemize}
                    \item
                        Code for rendering the Barnsley fern and variants.
                    \item
                        Implementations: \texttt{C, C++, Python, Go}
                \end{itemize}
            \item
                \href{%
                    https://github.com/ryanmaguire/complex_visual_plots/%
                }{\texttt{complex\_visual\_plots}}
                \begin{itemize}
                    \item
                        Tools for drawing plots of complex functions. Capable
                        of visualizing Newton and Mandelbrot iterates and their
                        fractal limits.
                    \item
                        Implementations: \texttt{C, C++, Go}
                \end{itemize}
            \item
                \href{%
                    https://github.com/ryanmaguire/mandelbrot_set/%
                }{\texttt{mandelbrot\_set}}
                \begin{itemize}
                    \item
                        Renderings of the Mandelbrot set and its variants.
                    \item
                        Implementations: \texttt{C, Python}
                \end{itemize}
            \item
                \href{%
                    https://github.com/ryanmaguire/metric_space_disks/%
                }{\texttt{metric\_space\_disks}}
                \begin{itemize}
                    \item
                        Draws the closed unit ball for arbitrary metrics in the
                        plane. The $L_{1}$, $L_{2}$, and $L_{\infty}$ metrics
                        are provided as examples.
                    \item
                        Implementations: \texttt{C}
                \end{itemize}
            \item
                \href{%
                    https://github.com/ryanmaguire/newton_fractals/%
                }{\texttt{newton\_fractals}}
                \begin{itemize}
                    \item
                        Renders Newton fractals for arbitrary polynomials
                        $p\in\mathbb{C}[z]$. Contains a full implementation of
                        the Hubbard--Schleicher--Sutherland algorithm for
                        finding all of the roots of a complex polynomial.
                    \item
                        Implementations: \texttt{C, Python}
                \end{itemize}
            \item
                \href{%
                    https://github.com/ryanmaguire/newtonian_black_holes/%
                }{\texttt{newtonian\_black\_holes}}
                \begin{itemize}
                    \item
                        Ray traces black holes by abusing Newtonian mechanics.
                        The force of gravity is allowed to act on light as if
                        photons had an arbitrarily small non-zero mass.
                        Numerical methods are used to trace the paths of the
                        light rays.
                    \item
                        Implementations:
                        \texttt{%
                            C, C++, Fortran, Go, IDL, Java,
                            Pascal, Python, Rust, Swift%
                        }
                \end{itemize}
            \item
                \href{%
                    https://github.com/ryanmaguire/perfectly_normal_spaces/%
                }{\texttt{perfectly\_normal\_spaces}}
                \begin{itemize}
                    \item
                        Visualization of the fact that $\mathbb{R}^{2}$ is a
                        perfectly normal topological space. Given any two
                        disjoint closed regions $\mathcal{C}$ and
                        $\mathcal{D}$, each a finite union of polygonal
                        figures, this plots the separating function
                        \[
                            f\big((x,\,y)\big)
                            =
                            \frac{%
                                \textrm{dist}\big((x,\,y),\,\mathcal{C}\big)%
                            }{%
                                \textrm{dist}\big((x,\,y),\,\mathcal{C}\big)%
                                +%
                                \textrm{dist}\big((x,\,y),\,\mathcal{D}\big)%
                            }.
                        \]
                        The provided example uses the Chinese characters
                        \textit{tu} and \textit{dou}, which together translate
                        to \textit{potato} in English, a running joke in my
                        topology class during the summer of 2022.
                    \item
                        Implementations: \texttt{C}
                \end{itemize}
            \item
                \href{%
                    https://github.com/ryanmaguire/random_walks/%
                }{\texttt{random\_walks}}
                \begin{itemize}
                    \item
                        Draws random walks on various manifolds. Currently
                        supports the torus, the Klein bottle, and the real
                        projective plane.
                    \item Implementations: \texttt{C, Asymptote}
                \end{itemize}
        \end{itemize}
